▌ 代码思路(伪代码)

  void draw_ending_A_rooftop(void) {
      // 1. 加载章 3 的天台场景
      loadScene("rooftop.txt");  // 已有函数 · 会按章 3 配色画
      // 注意:此时用的是 chapter=3 的配色 · 需要临时改成结局 A 配色

      // 2. 抹掉章 3 的物品标记
      gotoxy(20, 11); printf(" ");           // 抹 A
      gotoxy(32, 14); printf(" ");           // 抹 B
      gotoxy(24, 16); setColor(8);
      printf("~~");                           // C? → ~~
      for (int x = 5; x <= 44; x += 5) {     // 抹底部路径
          gotoxy(x, 19); printf(" ");
      }

      // 3. 加织女星
      gotoxy(44, 0); setColor(14); printf("✦");

      // 4. 独白分隔线
      setColor(8);
      gotoxy(0, 18);
      for (int i = 0; i < 49; i++) printf("═");

      // 5. 主角初始位置
      gotoxy(28, 14); setColor(15); printf("@");
  }

  // 织女星呼吸(和主界面一样)
  void vega_breathe(int x, int y) {
      int c[5] = {8, 7, 6, 14, 6};
      static int frame = 0;
      gotoxy(x, y); setColor(c[frame]); printf("✦");
      frame = (frame + 1) % 5;
  }

  // 手电光斑
  void flashlight_scan(void) {
      int path[3][2] = { {22,15}, {23,15}, {24,15} };
      for (int i = 0; i < 3; i++) {
          if (i > 0) {  // 擦除上一个
              gotoxy(path[i-1][0], path[i-1][1]);
              setColor(8); printf("~");
          }
          gotoxy(path[i][0], path[i][1]);
          setColor(14); printf("*");
          Sleep(300);
      }
  }
