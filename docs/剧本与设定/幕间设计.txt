============================================================
《星空回响》幕间旁白系统 · 视觉与分页设计
============================================================

【本文件说明】
- 定稿版：7 段幕间旁白 + 章节过场的完整设计
- 触发规则：第一次从 hall 进该场景时触发；开篇一段只在标题→hall 时触发
- 每段幕间：6 页画面（interlude[0] 8 页）
- 画面结构：上半屏 20×48 场景图 + 下半屏 8 行文字
- 每页通过"颜色高亮"点亮画面上不同的物件


============================================================
【一、整体架构】
============================================================

▌ 三层触发：

  开篇 interlude[0]：只在【标题→hall】播 1 次
    - 无章节过场
    - 8 页作者前言 + 书房静物

  章节过场 chapter_transitions[1..6]：每次【hall→场景】都播
    - 全屏黑底 + ═══ 装饰线
    - 编号 + 章节名 + 短旁白（骨架）
    - ~7.5 秒

  幕间旁白 interlude[1..6]：只在【第一次 hall→场景】播 1 次
    - 6 页画面 + 长散文（血肉）
    - ~30 秒

▌ 完整流程：

  第一次进章 N：
    hall 按 E → 章节过场 (~7.5s) → 幕间散文 (~30s) → 加载场景

  第二次+ 进章 N：
    hall 按 E → 章节过场 (~7.5s) → [幕间跳过] → 加载场景

  Q 回 hall：
    不播任何东西，直接回 hall

  开篇：
    标题按空格 → interlude[0] → 加载 hall

【十三、核心视觉命题】
============================================================

★ ⭐ 是主线物件
  从"未知的星"（不知谁折的）
  到"展开的字"（她的字），
  代表：她一直在等他准备好。

★ 织女星 ✦ 是她的化身
  全篇位置从不变，只有亮度在变。
  interlude[4] 唯一一次暗掉 ——
  那是他一生里最不敢抬头的时刻。

★ 从暖开始 → 越走越冷 → 最后回到暖
  这是整个故事的形状。

// 幕间画面（每段一张 20×48 底图）
const char* interlude_bg[7][20];

// 每一页的：文字 + 高亮列表
typedef struct {
    const wchar_t* text;          // 该页文字
    int highlight[8][3];          // 最多 7 处 {x, y, color}，末尾 {-1,-1,-1}
} interlude_page;

const interlude_page interlude_pages_0[8];   // 开篇 8 页
const interlude_page interlude_pages_1[6];   // 序幕 6 页
const interlude_page interlude_pages_2[6];   // 图书馆
const interlude_page interlude_pages_3[6];   // 天台
const interlude_page interlude_pages_4[6];   // 医院
const interlude_page interlude_pages_5[6];   // 天文馆
const interlude_page interlude_pages_6[6];   // 回响

const interlude_page* interlude_pages[7];
const int interlude_page_count[7] = { 8, 6, 6, 6, 6, 6, 6 };

// 每段是否播过
int interlude_played[7] = {0};

// 章节过场
typedef struct {
    const wchar_t* label;
    const wchar_t* name;
    const wchar_t* line1;
    const wchar_t* line2;
} ChapterTransition;

const ChapterTransition chapter_transitions[7];

【十五、播放函数（大纲）】
============================================================

// 章节过场（照搬原策划时序）
void play_chapter_transition(int chapter);

// 幕间旁白（含 6 页画面 + 分页文字 + 高亮）
void play_interlude(int idx) {
    if (interlude_played[idx]) return;
    for (each page in interlude_pages[idx]) {
        1. system("cls")
        2. draw_interlude_bg(idx)                   底图全灰
        3. apply_highlight(idx, page.highlight)      改色/新增
        4. gotoxy(2, 14); setColor(15);
           delay_with_Chinese(page.text)             打字机
        5. gotoxy(2, 22); "—— 按 Enter 继续 ——"
        6. while (_getch() != '\r') {}
    }
    interlude_played[idx] = 1;
}
