============================================================
《星空回响》对话框设计文档
============================================================

【适用范围】
- 主线：物品交互对话
- Vega 线：触发点独白（姓名栏填 "Vega"，浅紫色）
- 结局 / TE 的居中大段文字：不用本对话框，见"结局与TE画面稿"

【重要排版原则】
- 所有含中文的对话框：只画"上下横线"，不画左右竖线
- 这样 VSCode 里怎么看都对齐，运行时用 gotoxy 精确定位
- 左右竖线由程序在运行时根据"最长中文行"动态绘制

一、整屏布局（80 列 × 22 行）
============================================================

  行 00  ：[HUD]  记忆: 87 / 100                  第 2 章 · 相 伴
  行 01  ：（空）
  行 02  ：（空）
     ...
  行 21  ：（20 行场景对应 20 行右侧对话区）
  行 22  ：[提示条]  SPACE 继续    B 回看     ESC 菜单

  列位分配：
      col  1..48 : 场景（现有 20×48 ASCII）
      col 49..50 : 间隔（2 个空格）
      col 51..80 : 对话框（30 列）
    → 总宽 80 列，兼容 cmd 默认大小。

  说明：不激活对话时，col 51..80 整片留空。


============================================================
二、对话框结构（视觉设计）
============================================================

对话框：上下横线 + 中间中文自由排列（无右竖线）

  --------------------------------
   ⭐ 折纸星星                          ← 姓名栏（金色）
  --------------------------------

     你拿起那颗星星。
     它很旧了，
     边角都卷起来了。                   ← 文本区（最多 10 行）
     你不记得是谁折的。
     但你的手指还记得怎么把它
     拆开。

                              ▼        ← 继续提示（闪烁）
  --------------------------------

  ★ 排版规则：
     - 上下横线：30 个 '-'（纯 ASCII，永远对齐）
     - 姓名栏：距左 1 空格开始
     - 文本区：距左 5 空格开始
     - ▼：距左 30 空格（贴近横线右端）
     - 中间没有 '|' 竖线，中文再多都不会歪

  ★ 定位：起点在 (行 3, 列 51)

============================================================
四、文本渲染规则
============================================================

【分段】
- 剧本每换一段（空行分隔）= 打字机一次
- 一段播完 → 右下角 ▼ 出现（每 500 ms 闪烁）
- 按 SPACE → 清空文本区 → 播下一段
- 全部段播完 → 按 SPACE → 关闭对话框

【打字机】
- 默认速度：40 ms / 字
- 可在 ESC 菜单调：30 / 40 / 50 / 70 / 100 ms
- ★ 播放中按 SPACE 无效 ★（本项目不做快进）
- 播放中 ▼ 不显示

【换行】
- 剧本源文件里已经写好断行
- 一行不超过 12 个汉字
- 一段超过 10 行时在剧本里手动拆两段

【新段处理】
- 清空文本区（左上角开始）
- 从头播新段
- 不做累积滚动


============================================================
五、按键说明
============================================================

  对话框激活期间：

    SPACE      打字机播放中 → 无效
               本段已播完   → 下一段 / 关闭对话框
    B          回看历史（弹出对话历史栈的最近 5 条）
    ESC        打开菜单（暂停）
    W/A/S/D    无效（对话时禁止移动）
    E          无效


============================================================
六、状态时机
============================================================

【对话打开时】
  ★ 记忆值倒计时暂停 ★    ← 关键，避免读长文触发结局 C
  场景中的光标停在原位不闪
  场景物品字符继续显示（不被覆盖）

【对话关闭时】
  记忆值倒计时恢复
  若刚播完的是物品对话 → item.taken = true
                       → 该物品字符从场景抹除（下一帧重绘）
                       → 物品加入链表 / 对话入栈

【栈与链表】
  每次对话开始时       → 对话栈 push 一整段（用于 B 回看）
  每次物品拾取时       → 物品清单链表 append 一个节点

八、实现建议（数据结构骨架）
============================================================

// 一段对话（一次打字机播放）
struct Segment {
    const char *text;    // 完整文字，内部可含 '\n' 断行
};

// 一个物品的对话数据
struct Dialog {
    const char *name;    // "⭐ 折纸星星" / "Vega" / ...
    int   nameColor;     // 14 金 / 13 浅紫
    int   segCount;      // 段数
    Segment segs[8];     // 最多 8 段
};

// 运行时状态
Dialog *cur       = NULL;
int     segIdx    = 0;      // 当前段号
int     charIdx   = 0;      // 打字机进度（本段第几字）
bool    segDone   = false;  // 本段是否播完
DWORD   lastChar  = 0;      // 上次出字的时刻

// 画对话框骨架（只画上下横线，不画竖线）
void drawDialogFrame(int x, int y, int width) {
    gotoxy(x, y);
    for (int i = 0; i < width; i++) putchar('-');
    // 姓名栏在 y+1
    // 分隔线在 y+2
    gotoxy(x, y + 2);
    for (int i = 0; i < width; i++) putchar('-');
    // 下横线在 y+14
    gotoxy(x, y + 14);
    for (int i = 0; i < width; i++) putchar('-');
}

// 主循环里的对话更新
void updateDialog() {
    if (!cur) return;

    // 打字机
    if (!segDone) {
        DWORD now = GetTickCount();
        if (now - lastChar >= typewriter_ms) {
            charIdx++;
            lastChar = now;
            if (reachedEnd(cur->segs[segIdx].text, charIdx))
                segDone = true;
        }
    }

    // 按键
    if (segDone && spacePressed()) {
        segIdx++;
        if (segIdx >= cur->segCount) {
            closeDialog();          // 结束
        } else {
            charIdx = 0;
            segDone = false;
            lastChar = GetTickCount();
            clearTextArea();        // 清空文本区
        }
    }
}
