【六、代码实现思路】
============================================================

▌ 通用献词函数

  void show_dedication(int ending_type) {
      // ending_type: 0=A, 1=B, 2=C
      system("cls");

      // 1. ✦ 亮起
      int vega_x = 24, vega_y = 9;
      if (ending_type == 0) {
          gotoxy(vega_x, vega_y); setColor(14); printf("✦");
      } else if (ending_type == 1) {
          gotoxy(vega_x, vega_y); setColor(8); printf("✦");
      } else {
          gotoxy(vega_x, vega_y); setColor(1); printf("✦");
      }
      Sleep(1000);

      // 2. 逐行显示献词文字
      const wchar_t* lines_A[] = {
          L"献给她。",
          L"献给三年那个不能回去的夕阳。",
          L"献给每一个抬头看星星的人。",
          L"你做到了。",
          NULL
      };
      const wchar_t* lines_B[] = {
          L"献给那些今夜也睡不着的人。",
          L"献给那些不是因为想过、",
          L"而是因为得过的人。",
          L"献给拆了又折回去的人。",
          L"到了明天",
          L"你还会把那颗星星、折回去。",
          NULL
      };
      const wchar_t* lines_C[] = {
          L"献给那些已经忘记的人。",
          L"献给那些记得一切、",
          L"却不知道为什么重要的人。",
          L"献给她。",
          L"—— 即使你已经不记得她是谁。",
          NULL
      };

      const wchar_t** lines;
      int line_count;
      if (ending_type == 0) { lines = lines_A; line_count = 4; }
      else if (ending_type == 1) { lines = lines_B; line_count = 6; }
      else { lines = lines_C; line_count = 5; }

      for (int i = 0; i < line_count; i++) {
          int len = (int)wcslen(lines[i]);
          int x = (50 - len * 2) / 2;
          int y = 11 + i;
          int color = 15;
          // A 的最后一行用金色
          if (ending_type == 0 && i == line_count - 1) color = 14;
          // C 的"献给她"用白 15(短促亮)
          if (ending_type == 2 && i == line_count - 2) color = 15;
          // C 的"即使你已经不记得"用灰 7
          if (ending_type == 2 && i == line_count - 1) color = 7;

          gotoxy(x, y); setColor(color);
          printWideAll(lines[i]);
          Sleep(200);
      }

      // 3. 作品名
      Sleep(400);
      int title_x = (50 - 12) / 2;  // "《星空回响》" 6 个中文字符 = 12 列
      gotoxy(title_x, 20); setColor(8);
      printWideAll(L"《星空回响》");

      // 4. 按键提示
      Sleep(300);
      gotoxy(18, 24); setColor(8);
      printf(" [ 按 空 格 继 续 ]");

      // 5. 等待按键
      clearbutter();
      _getch();
  }

【设计备注】
============================================================

▌ A 的献词 · 温度
   "献给她。" 开头 · 和 A 结局最后的"她做到了"呼应
   "你做到了。" 用金色 · 呼应织女星 · 呼应"我想成为你的牛郎星"

▌ B 的献词 · 无奈
   "献给那些不是因为想过、而是因为得过的人。" 是 B 的核心句
   "到了明天你还会把那颗星星折回去。" 呼应 B 段 5 的折回去动作
   全篇用灰 7 · 冷色到底

▌ C 的献词 · 最重
   "献给她。—— 即使你已经不记得她是谁。" 是 C 的最后一击
   "献给她。" 用白 15 短暂亮一下 · 然后下一句立刻变灰
   —— 表达"她"这个字短暂亮过 · 然后立刻熄灭
   ✦ 从暗 1 到灰 8 · 永远达不到金色

▌ 三个献词 · 三个语气
   A 是"你做到了" —— 释怀 · 温暖 · 抵达
   B 是"你还会折回去" —— 无奈 · 继续 · 从中
   C 是"即使你已经不记得" —— 遗忘 · 沉重 · 归零
