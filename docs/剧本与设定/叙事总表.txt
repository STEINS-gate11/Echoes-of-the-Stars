============================================================
《星空回响》叙事总表（对话 / 旁白 / 三重结局 / 记忆值系统）
============================================================

【本文件说明】
- 汇总全部叙事资产：15 段物品对话 + 5 段层间旁白 + 3 种结局 + 记忆值机制
- 是 物品对话.txt 与 结局与记忆值.txt 的合并升级版
- 编码建议 GBK（Windows 控制台默认）；如用 UTF-8 需 chcp 65001
- 打字机效果：文本按行输出，每字符 30-50ms 延迟  

【玩家操作复习】
- WASD  移动光标
- E     与物品交互（触发对话 → 拾取 → 加入链表 → 压入对话栈）
- B     回看对话历史（栈）
- C     查看已收集物品清单（链表）
- ESC   暂停 / 退出

【第三部分 · 记忆值系统】
============================================================

▌ 初始值   ：100
▌ 衰减     ：每 8 秒 -1
▌ 归零总时 ：8 × 100 = 800 秒 ≈ 13 分 20 秒
▌ HUD 显示 ：屏幕右上角 "记忆: 87 / 100"
▌ 颜色分级 ：
    · 100-60 → 白色（安全）
    · 59 -30 → 黄色（提醒该赶路了）
    · 29 -1  → 红色 + 闪烁（危险）
▌ 停止条件 ：
    · 玩家进入 finale 并触发 A / B → 停止倒计时
    · 记忆归零瞬间未触发 A / B → 强制进入结局 C
▌ 玩法意义 ：
    "记忆是会流失的东西。"
    时间压力不是惩罚探索慢，而是让"重新想起"这件事
    本身带上分量——你必须赶在忘掉之前，把她想起来。

// ---------- 物品数据 ----------
struct Item {
    char anchor;         // 'A' / 'B' / 'C'
    int x, y;
    const char *emoji;   // "⭐" "📖" ...
    const char *name;    // "折纸星星"
    const char *dialog;  // 完整对话文本（可含 \n）
    bool hidden;         // 是否隐藏物品
    bool taken;          // 是否已拾取
};

// 6 层，每层 3 件
Item scenes[6][3] = {
    // 序幕
    {

       "你拿起那颗星星。\n它很旧了，边角都卷了起来。\n"
       "你不记得是谁折的,\n但你的手指还记得怎么把它拆开。",
       false, false},
      {'B', 25, 12, "📖", "空白日记本",
       "封面写着：\"给未来的你。\"\n里面一页都没写过。\n...",
       false, false},
      {'C', 10, 17, "🖼", "模糊照片",
       "照片很模糊...", true, false}
    },
    // 图书馆 / 天台 / 医院 / 天文馆 / 回响 依次填入
    ...
};

// 层间旁白
const char *transitions[5] = {
    "你不记得她是谁了。\n但你还记得，你答应她过要去看星星。",
    "后来你们真的去看了很多次星星。\n那时候你以为，这只是个开始。",
    "然后有一天，你发现星星也会消失。\n你只是没想到，这一次会轮到她。",
    "她没有怪你。\n但你怪了你自己很久。",
    "你终于准备好了。"
};


// ---------- 记忆值倒计时 ----------
int memory = 100;                    // 当前记忆值 (0..100)
DWORD lastTick = 0;                  // 上次扣减时刻 (ms)
const DWORD TICK_MS = 8000;          // 8 秒扣 1 点
bool inFinaleEnding = false;         // 已进入 finale 结局判定?

enum Ending { ENDING_NONE, ENDING_A, ENDING_B, ENDING_C };

void initMemory() {
    memory = 100;
    lastTick = GetTickCount();
    inFinaleEnding = false;
}

Ending updateMemory() {
    if (inFinaleEnding) return ENDING_NONE;
    DWORD now = GetTickCount();
    while (now - lastTick >= TICK_MS && memory > 0) {
        memory--;
        lastTick += TICK_MS;
    }
    if (memory <= 0) {
        inFinaleEnding = true;
        return ENDING_C;
    }
    return ENDING_NONE;
}

int memoryColor(int m) {
    if (m >= 60) return 15;   // 白
    if (m >= 30) return 14;   // 黄
    return 12;                 // 红（<30 可再叠加闪烁）
}

void drawMemoryHUD() {
    gotoxy(36, 0);
    setColor(memoryColor(memory));
    printf("记忆: %3d / 100", memory);
    setColor(15);
}


// ---------- 结局选择 ----------
Ending chooseFinaleEnding(int totalTaken, int hiddenMissed) {
    inFinaleEnding = true;                   // 停止倒计时
    if (totalTaken >= 15)  return ENDING_A;
    if (hiddenMissed >= 2) return ENDING_B;
    return ENDING_B;
}

// 主循环骨架
void gameLoop() {
    initMemory();
    while (true) {
        Ending e = updateMemory();
        if (e == ENDING_C) { playEndingC(); break; }

        handleInput();
        renderScene();
        drawMemoryHUD();

        if (playerReachedFinale()) {
            int total = countAllTaken();
            int miss  = countMissedHidden();
            Ending fe = chooseFinaleEnding(total, miss);
            if (fe == ENDING_A) playEndingA();  // 含"她说"追加段落
            else                playEndingB();
            break;
        }
        Sleep(30);
    }
}

【第六部分 · 调优建议】
============================================================

▌ 800 秒对不同玩家的评估：
    · 通关型（直奔物品）  → 时间充裕
    · 品味型（细读对话）  → 偏紧
    · 探索型（每个角落走）→ 可能触发结局 C

▌ 若测试后觉得太紧，改一个常量即可：
    const DWORD TICK_MS = 10000;  // 10 秒/点 → 1000 秒 ≈ 16.7 分
    const DWORD TICK_MS = 12000;  // 12 秒/点 → 1200 秒 = 20 分

▌ 可选加分项："补给"机制
    · 拾取隐藏物品(C) 回复 +5 记忆值
    · 触发层间旁白 回复 +3 记忆值
    · 让玩家有"我在往回想"的正反馈

▌ 结局 C 出现时机的两个设计选择：
    · 硬切断：memory 归零瞬间立刻跳结局 C（简单）
    · 缓冲期：归零后再给 15 秒"最后回光"，画面开始
      抖动/褪色，玩家最后一次交互仍算数——若在这
      15 秒内跑到 finale 仍可救回结局 B（更有戏剧性）
