============================================================
第一部分：整体架构（俯视图）
============================================================

游戏本质上是一个"每 30 毫秒问自己一次'现在是谁'的循环"。

    ┌─────────────────────────────────────────────┐
    │                                             │
    │   ┌────────┐   ┌────────┐   ┌────────┐    │
    │   │ INPUT  │→│ UPDATE │→│ RENDER │→ Sleep │
    │   └────────┘   └────────┘   └────────┘    │
    │        ↑                        │           │
    │        └────────────────────────┘           │
    │             一个循环 = 一帧                  │
    │                                             │
    └─────────────────────────────────────────────┘

每一帧只做三件事：
  1. 读键盘 (input)
  2. 更新状态 (update)
  3. 画屏幕 (render)

"状态"包括：现在在哪个场景、光标在哪、记忆值多少、
对话是否激活、结局有没有触发……

一切复杂性都藏在"状态"里。


============================================================
第二部分：分层设计（三层）
============================================================

┌───────────────────────────────────────────────────────┐
│  第 3 层 · 表现层 (Presentation)                       │
│      · 控制台绘制（gotoxy / setColor / printf）        │
│      · 场景 .txt 加载 / 打字机 / 淡入淡出              │
│      · BGM / 音效播放                                  │
├───────────────────────────────────────────────────────┤
│  第 2 层 · 逻辑层 (Logic)                              │
│      · 状态机（标题 → 场景 → 结局）                    │
│      · 碰撞检测 / 交互判定 / 结局触发                  │
│      · 记忆值倒计时 / 存档读写                         │
├───────────────────────────────────────────────────────┤
│  第 1 层 · 数据层 (Data)                               │
│      · 场景背景数组 char bg[20][49]                    │
│      · 物品数组 Item scenes[6][3]                      │
│      · 对话文本 / 旁白 / 结局文字（const 表）          │
│      · 存档结构 SaveData                               │
└───────────────────────────────────────────────────────┘

★ 数据层是静态的（const，编译时确定），逻辑层调用它。
★ 逻辑层不直接调 printf，它调表现层。
★ 表现层不知道"当前在哪个场景"，它只画你给它的东西。

第四部分：状态机（核心）
============================================================

整个游戏就是一个大的状态机。

    enum GameState {
        STATE_TITLE,           // 标题界面
        STATE_CHAPTER_INTRO,   // 章节过场（第 X 章）
        STATE_EXPLORE,         // 场景探索（WASD 走位）
        STATE_DIALOG,          // 对话框激活中
        STATE_TRANSITION,      // 层间旁白播放
        STATE_ENDING_A,        // 结局 A 播放
        STATE_ENDING_B,        // 结局 B 播放
        STATE_ENDING_C,        // 结局 C 播放
        STATE_VEGA_OPENING,    // Vega 开场引言
        STATE_VEGA_EXPLORE,    // Vega 场景走位
        STATE_VEGA_MONO,       // Vega 独白播放
        STATE_VEGA_ENDING,     // Vega 结尾字幕
        STATE_TE,              // TE 全程播放
        STATE_QUIT             // 退出游戏
    };

主循环：

    while (state != STATE_QUIT) {
        handleInput();
        updateState();       // 里面根据 state 分派
        renderState();       // 里面根据 state 分派
        Sleep(30);
    }

updateState() 里：
    switch (state) {
        case STATE_TITLE:      updateTitle();      break;
        case STATE_EXPLORE:    updateExplore();    break;
        case STATE_DIALOG:     updateDialog();     break;
        ...
    }

renderState() 类似。

★ 每个状态一个小函数，各管各的。
★ 状态切换只在 update 里发生（比如玩家按 E 触发对话 → state = STATE_DIALOG）

★ 静态数据用 const，逻辑层只读。
★ 运行时数据是全局变量（用一个 struct 包起来更好）。

    int memory = 100;
    DWORD lastTick;
    bool paused;

    updateMemory():
      if (paused) return;
      DWORD now = GetTickCount();
      while (now - lastTick >= 8000 && memory > 0) {
          memory--;
          lastTick += 8000;
      }
      if (memory <= 0 && !inFinaleEnding) {
          state = STATE_ENDING_C;
      }

    drawHUD():
      gotoxy(60, 0);
      setColor(memoryColor(memory));
      printf("记忆: %d / 100", memory);

  对话激活时 memoryPaused = true
  对话关闭时 memoryPaused = false + lastTick = GetTickCount()

第九部分：存档系统（很简单）
============================================================

  save/save.cpp:
    struct SaveData {
        bool endingA_cleared;
        bool endingB_cleared;
        bool endingC_cleared;
        bool vega_cleared;
        bool te_cleared;
    };

    void loadSave(SaveData *s):
      FILE *fp = fopen("save.dat", "rb");
      if (fp) { fread(s, sizeof(SaveData), 1, fp); fclose(fp); }
      else    { memset(s, 0, sizeof(SaveData)); }

    void writeSave(const SaveData *s):
      FILE *fp = fopen("save.dat", "wb");
      fwrite(s, sizeof(SaveData), 1, fp);
      fclose(fp);

  游戏启动时 loadSave 一次
  任一结局触发时改 s 里对应字段 + writeSave

第十部分：BGM 系统
============================================================

  audio/audio.cpp:
    #include <windows.h>
    #pragma comment(lib, "winmm.lib")

    static char currentBGM[256] = "";

    void playBGM(const char *file):
      if (strcmp(currentBGM, file) == 0) return;  // 同一首不切
      if (currentBGM[0]) {
          mciSendString("stop bgm", 0, 0, 0);
          mciSendString("close bgm", 0, 0, 0);
      }
      char cmd[256];
      sprintf(cmd, "open \"%s\" alias bgm", file);
      mciSendString(cmd, 0, 0, 0);
      mciSendString("play bgm repeat", 0, 0, 0);
      strcpy(currentBGM, file);

    void stopBGM():
      mciSendString("stop bgm", 0, 0, 0);
      mciSendString("close bgm", 0, 0, 0);
      currentBGM[0] = 0;

    void setBGMVolume(int vol):  // 0-1000
      char cmd[64];
      sprintf(cmd, "setaudio bgm volume to %d", vol);
      mciSendString(cmd, 0, 0, 0);

  淡入淡出：用 timer 循环调 setBGMVolume。

第十一部分：主循环最终形态（伪代码）

int main() {
    // 启动初始化
    hideCursor();
    setConsoleUTF8();
    loadSave(&save);
    audio.playBGM("res/bgm/title.mp3");  // 标题 BGM

    GameState state = STATE_TITLE;

    while (state != STATE_QUIT) {
        // 1. 输入
        char key = readKey();

        // 2. 更新
        updateMemory();  // 除非 paused

        switch (state) {
            case STATE_TITLE:      updateTitle(key);      break;
            case STATE_CHAPTER_INTRO: updateChapter(key); break;
            case STATE_EXPLORE:    updateExplore(key);    break;
            case STATE_DIALOG:     updateDialog(key);     break;
            case STATE_ENDING_A:   updateEndingA(key);    break;
            // ... 其他状态
        }

        // 3. 渲染
        renderCurrentState();

        // 4. 睡眠
        Sleep(30);
    }

    // 收尾
    audio.stopBGM();
    return 0;
}
